zapytanie 2: SELECT obiekty.plik, obiekty.nazwa, obiekty.nazwaCechy, obiekty.wartoscCechy, obiekty.opis, rodzaje.rodzaj FROM `obiekty` JOIN rodzaje ON obiekty.idRodzaj=rodzaje.idRodzaj;
